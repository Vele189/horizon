# What Streamlit Community Cloud installs.
#
# This file is the deployed dashboard's runtime and nothing else. Everything
# that builds the warehouse the dashboard reads — the API client, dbt, the
# model — lives in requirements-pipeline.txt, because Cloud installs *this*
# file on every deploy and none of that is imported by a page. Shipping it
# anyway cost a 1.5 GB install to serve an app that imports three libraries,
# and put dbt-core's transitive pins on the critical path of a deploy that has
# nothing to do with dbt.
#
#     pip install -r requirements.txt            # run the dashboard
#     pip install -r requirements-pipeline.txt   # build what it reads
#     pip install -r requirements-dev.txt        # run the tests
#
# The split only holds if it is checked: tests/test_dashboard.py walks every
# import under dashboard/ and fails if one of them is missing here. A chart
# added in BI-03 cannot reach Cloud with its dependency left behind.

streamlit==1.63.0
pandas==3.0.5
SQLAlchemy==2.0.52

# pandas requires numpy regardless; pinned so the deployment resolves to the
# same version the marts were built and the model was trained against.
numpy==2.5.3

# The driver SQLAlchemy loads from a postgresql:// URL. Nothing imports it by
# name, which is precisely why it is named here rather than left to the import
# scan to notice — a dashboard with no driver fails at first query, in front
# of a visitor, with a message about a missing dialect.
psycopg2-binary==2.9.10

# config.py is the only module in the project that reads the environment, and
# the dashboard goes through it for the local fallback, so its one dependency
# comes along.
python-dotenv==1.0.1
